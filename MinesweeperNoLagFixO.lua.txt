local function N(cls, p)
    local o = Instance.new(cls)
    for k, v in pairs(p) do
        if k ~= "Par" then
            pcall(function() o[k] = v end)
        end
    end
    if p.Par then o.Parent = p.Par end
    return o
end

local function Crn(r)
    return N("UICorner", { CornerRadius = UDim.new(0, r), Par = nil })
end

local function getGui()
    local name = "MoonUIv2"
    -- clean up old instances
    pcall(function()
        local cg = game:GetService("CoreGui")
        local old = cg:FindFirstChild(name)
        if old then old:Destroy() end
    end)
    pcall(function()
        local pg = LP:FindFirstChild("PlayerGui")
        if pg then
            local old = pg:FindFirstChild(name)
            if old then old:Destroy() end
        end
    end)
    if gethui then
        pcall(function()
            local old = gethui():FindFirstChild(name)
            if old then old:Destroy() end
        end)
    end

    local sg = Instance.new("ScreenGui")
    sg.Name          = name
    sg.ResetOnSpawn  = false
    sg.DisplayOrder  = 999
    pcall(function() sg.ZIndexBehavior = Enum.ZIndexBehavior.Sibling end)
    pcall(function() sg.IgnoreGuiInset = true end)

    if gethui then
        sg.Parent = gethui()
    else
        local ok = pcall(function() sg.Parent = game:GetService("CoreGui") end)
        if not ok then
            local pg = LP:FindFirstChild("PlayerGui") or LP:WaitForChild("PlayerGui", 5)
            if pg then sg.Parent = pg end
        end
    end
    return sg
end

-- Simple drag: attach to any Frame, moves target
-- Touch: locks to the first finger only, ignores additional fingers
local function makeDraggable(handle, target, blockList)
    local dragging    = false
    local startMouse  = Vector2.zero
    local startPos    = Vector2.zero
    local touchId     = nil  -- KeyCode of the locked finger

    handle.InputBegan:Connect(function(i)
        -- Mouse: only left button; Touch: only if no finger locked yet
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            -- blocklist check
            if blockList then
                local mp = UserInputService:GetMouseLocation()
                for _, b in ipairs(blockList) do
                    local ap, as = b.AbsolutePosition, b.AbsoluteSize
                    if mp.X >= ap.X and mp.X <= ap.X + as.X
                    and mp.Y >= ap.Y and mp.Y <= ap.Y + as.Y then return end
                end
            end
            dragging  = true
            touchId   = nil
            startMouse = UserInputService:GetMouseLocation()
        elseif i.UserInputType == Enum.UserInputType.Touch and touchId == nil then
            -- lock to this finger only
            if blockList then
                local mp = Vector2.new(i.Position.X, i.Position.Y)
                for _, b in ipairs(blockList) do
                    local ap, as = b.AbsolutePosition, b.AbsoluteSize
                    if mp.X >= ap.X and mp.X <= ap.X + as.X
                    and mp.Y >= ap.Y and mp.Y <= ap.Y + as.Y then return end
                end
            end
            dragging  = true
            touchId   = i.KeyCode  -- unique per finger
            startMouse = Vector2.new(i.Position.X, i.Position.Y)
        else
            return
        end
        local parentAbs = target.Parent and target.Parent.AbsolutePosition or Vector2.zero
        local abs = target.AbsolutePosition
        startPos = Vector2.new(abs.X - parentAbs.X, abs.Y - parentAbs.Y)
    end)

    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        elseif i.UserInputType == Enum.UserInputType.Touch and i.KeyCode == touchId then
            dragging = false
            touchId  = nil
        end
    end)

    UserInputService.InputChanged:Connect(function(i)
        if not dragging then return end
        local m
        if i.UserInputType == Enum.UserInputType.MouseMovement then
            m = UserInputService:GetMouseLocation()
        elseif i.UserInputType == Enum.UserInputType.Touch and i.KeyCode == touchId then
            m = Vector2.new(i.Position.X, i.Position.Y)
        else
            return  -- ignore other fingers
        end
        local d = m - startMouse
        target.Position = UDim2.fromOffset(startPos.X + d.X, startPos.Y + d.Y)
    end)
end

-- Quick-press: reliable on mouse and touch
local function quickPress(btn, cb)
    local t0, startPos, moved = nil, nil, false
    btn.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            t0 = tick()
            startPos = Vector2.new(i.Position.X, i.Position.Y)
            moved = false
        end
    end)
    btn.InputChanged:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch and startPos then
            local d = Vector2.new(i.Position.X, i.Position.Y) - startPos
            if d.Magnitude > 8 then moved = true end
        end
    end)
    btn.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            if t0 and not moved and (tick() - t0) < 0.25 then
                cb()
            end
            t0, startPos, moved = nil, nil, false
        end
    end)
end

            Crn(6).Parent = thumb

            local sd = false
            local function upd(x)
                local r = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X,1), 0, 1)
                local nv = math.floor(mn + (r * (mx - mn)) + 0.5)
                if nv == val then return end

                if i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch then upd(i.Position.X) end
            end)
            UserInputService.InputEnded:Connect(function(i)
                if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then sd=false end
            end)

local abs, floor, huge, sqrt, max, min, tsort, tinsert, tremove = math.abs, math.floor, math.huge, math.sqrt, math.max, math.min, table.sort, table.insert, table.remove

local flagDelay = 0.1

local function scanB() local f = workspace:FindFirstChild("Flag") local p = f and f:FindFirstChild("Parts") if p then return p end for _, v in ipairs(workspace:GetChildren()) do if v:IsA("Folder") and #v:GetChildren() > 50 then return v end end return nil end
local function hasF(p) return p:FindFirstChildOfClass("Model") ~= nil end

local function applyHL(c, col) if not c.part or c.hlPart then return end local sg = inst("SurfaceGui") sg.Face = Enum.NormalId.Top sg.AlwaysOnTop = true sg.Parent = c.part local frame = inst("Frame") frame.Size = UDim2.fromScale(1,1) frame.BackgroundColor3 = col frame.BackgroundTransparency = 0.5 frame.Parent = sg c.hlPart = sg end
local function hideHL(c) if c.hlPart then c.hlPart:Destroy() c.hlPart = nil end end
local function clearAllHL() for x = 0, state.grid.w-1 do for z = 0, state.grid.h-1 do if state.cells.grid[x] and state.cells.grid[x][z] then hideHL(state.cells.grid[x][z]) end end end hlActive = {} end

local function cluster(vals)
    local res={} if #vals==0 then return res end local cur=vals[1]
    for i=2,#vals do if abs(vals[i]-cur)<4 then cur=(cur+vals[i])/2 else tinsert(res,cur) cur=vals[i] end end
    tinsert(res,cur) return res
end

local function rebuildG(folder)
    clearAllHL() for _, c in pairs(signalConns) do c:Disconnect() end signalConns = {} state.cells = { grid = {}, numbered = {}, toFlag = {}, toClear = {} }
    local pts = folder:GetChildren() if #pts == 0 then return end
    local xs, zs = {}, {} for i, p in ipairs(pts) do if p:IsA("BasePart") then xs[i]=p.Position.X zs[i]=p.Position.Z end end tsort(xs) tsort(zs)
    local ux, uz = cluster(xs), cluster(zs) state.grid.w, state.grid.h = #ux, #uz
    for x = 0, state.grid.w-1 do state.cells.grid[x] = {} end
    for _, p in ipairs(pts) do
        if p:IsA("BasePart") then
            local bx, bz = 1, 1 for i=1,#ux do if abs(p.Position.X - ux[i]) < 2 then bx = i break end end for i=1,#uz do if abs(p.Position.Z - uz[i]) < 2 then bz = i break end end
            local c = {part=p, state="unknown", number=0, covered=true, neigh={}, ix=bx-1, iz=bz-1}
            state.cells.grid[bx-1][bz-1] = c
            local conn = p.ChildAdded:Connect(function(child) if child.Name == "NumberGui" then c.state = "number" c.covered = false local lbl = child:FindFirstChild("TextLabel") if lbl then c.number = tonumber(lbl.Text) or 0 end hideHL(c) if not queueSet[c] then queueSet[c]=true tinsert(changeQueue, c) end solverDirty = true end end)
            table.insert(signalConns, conn)
        end
    end
    for x=0, state.grid.w-1 do for z=0, state.grid.h-1 do local c = state.cells.grid[x] and state.cells.grid[x][z] if c then for dx=-1,1 do for dz=-1,1 do if dx~=0 or dz~=0 then local n = state.cells.grid[x+dx] and state.cells.grid[x+dx][z+dz] if n then tinsert(c.neigh, n) end end end end end end end
    print("[Solver] Grid built: "..state.grid.w.."x"..state.grid.h)
end

local function processQueue()
    if #changeQueue == 0 then return end
    local batch = changeQueue changeQueue = {} queueSet = {}
    for _, c in ipairs(batch) do
        if c.part and c.state ~= "number" and c.state ~= "flagged" then
            local ng = c.part:FindFirstChild("NumberGui")
            if ng then
                c.state = "number" c.covered = false local lbl = ng:FindFirstChild("TextLabel") if lbl then c.number = tonumber(lbl.Text) or 0 end hideHL(c)
            end
        end
    end
    state.cells.numbered = {}
    for x=0, state.grid.w-1 do for z=0, state.grid.h-1 do local c = state.cells.grid[x] and state.cells.grid[x][z] if c and c.state=="number" then tinsert(state.cells.numbered, c) end end end
end

local function solveCSP(fS, sS)
    local num = state.cells.numbered
    for _, c in ipairs(num) do
        local unknown, flagged = 0, 0
        for _, n in ipairs(c.neigh) do if fS[n] or hasF(n.part) then flagged = flagged + 1 elseif n.covered then unknown = unknown + 1 end end
        local remaining = c.number - flagged
        if remaining == unknown and unknown > 0 then for _, n in ipairs(c.neigh) do if n.covered and not fS[n] then fS[n] = true end end end
        if remaining == 0 and unknown > 0 then for _, n in ipairs(c.neigh) do if n.covered and not sS[n] then sS[n] = true end end end
    end
end

local function solveLogic()
    if state.grid.w == 0 then return end
    local fS, sS = {}, {}
    processQueue()
    solveCSP(fS, sS)
    state.cells.toFlag, state.cells.toClear = fS, sS
end

local function updateVisuals()
    if not solverEnabled then return end
    for c in pairs(state.cells.toFlag) do if c.part and c.covered then applyHL(c, COLOR_MINE) end end
    for c in pairs(state.cells.toClear) do if c.part and c.covered then applyHL(c, COLOR_SAFE) end end
end
